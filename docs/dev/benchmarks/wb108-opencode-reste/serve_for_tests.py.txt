"""Test harness (WB-108): run the REAL board server in a subprocess.

The caller passes <port> and <home>. Inside this process:
  - HOME is already the temp home (env set by the test file), so the real
    load_config() reads the temp config.json exactly as in production.
  - The ticket store is the ONLY redirect: server.TICKETS_DIR is pointed at
    the temp home BEFORE the first request. Everything else — auth gate,
    session secret, login gate, Dispatcher, guard wiring, routing — is the
    unmodified module-level code of werkbank.server.
"""
import sys
import threading
import time
import urllib.request
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from werkbank import server  # noqa: E402  module-level wiring runs here, once


def main():
    port = int(sys.argv[1])
    home = Path(sys.argv[2])
    server.TICKETS_DIR = home / "tickets"

    class TestHandler(server.Handler):
        def do_GET(self):
            if self.path == "/__ready__":
                self._send(200, {"ok": True})
                return
            server.Handler.do_GET(self)

        def do_POST(self):
            if self.path == "/__ready__":
                self._send(200, {"ok": True})
                return
            server.Handler.do_POST(self)

    from http.server import ThreadingHTTPServer
    httpd = ThreadingHTTPServer(("127.0.0.1", port), TestHandler)

    def serve():
        httpd.serve_forever()

    t = threading.Thread(target=serve, daemon=True)
    t.start()

    last = None
    for _ in range(400):
        try:
            with urllib.request.urlopen(f"http://127.0.0.1:{port}/__ready__",
                                        timeout=1) as r:
                if r.status == 200:
                    print("READY", flush=True)
                    t.join()
                    return
        except Exception as e:
            last = e
            time.sleep(0.05)
    raise SystemExit(f"harness never became ready: {last!r}")


if __name__ == "__main__":
    main()
