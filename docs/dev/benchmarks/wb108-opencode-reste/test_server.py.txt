"""WB-108: the board's Auth-Gate, Guard and Routing as a COMPOSITION.

Every other test in this suite drives the pure functions (auth.verify_password,
guard.check_write, store.*). None of them ever asked what happens when a real
HTTP request reaches the real handler: is the auth check actually in the way?
is the guard wired into do_GET/do_POST? do the browse roots really come from
server.py, not from a copy of the logic?

Proven holes (2026-08-16): deleting the auth check in the handler, or widening
the browse roots to `/`, both left the whole pre-existing suite green. These
tests close the gap by running the UNMODIFIED Handler against a real
ThreadingHTTPServer on a free port (LAN mode + password) — so a regression in
the wiring itself is what fails the tests. The harness process uses an
isolated config and HOME (passed via env, because the board module is
imported once and its globals are process-wide), and the ticket store is
redirected to a temp dir via the only allowed seam: the harness sets the
module constant AFTER import, before any request is served.
"""
import json
import os
import shutil
import socket
import subprocess
import sys
import tempfile
import unittest
import urllib.error
import urllib.request
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent / "src"))

from werkbank import auth, store  # noqa: E402

REPO = Path(__file__).resolve().parent.parent
HARNESS_SCRIPT = REPO / "src" / "werkbank" / "serve_for_tests.py"


def _free_port() -> int:
    s = socket.socket()
    s.bind(("127.0.0.1", 0))
    port = s.getsockname()[1]
    s.close()
    return port


def _req(port, path, data=None, headers=None, method=None):
    """(status, raw_body, set_cookie) against the harness board."""
    body = None if data is None else (data if isinstance(data, bytes)
                                      else json.dumps(data).encode("utf-8"))
    req = urllib.request.Request(f"http://127.0.0.1:{port}{path}", data=body,
                                 method=method or ("POST" if body is not None else "GET"))
    for k, v in (headers or {}).items():
        req.add_header(k, v)
    try:
        with urllib.request.urlopen(req, timeout=30) as r:
            return r.status, r.read(), r.headers.get("Set-Cookie")
    except urllib.error.HTTPError as e:
        return e.code, e.read(), e.headers.get("Set-Cookie")


def _as_json(status, body):
    try:
        return json.loads(body.decode("utf-8"))
    except (ValueError, UnicodeDecodeError):
        return body


def board_headers(self, port: int = None) -> dict:
    port = port if port is not None else self.port
    return {"Content-Type": "application/json",
            "Host": f"{self.lan_ip}:{port}",
            "Origin": f"http://{self.lan_ip}:{port}"}


class BoardCompositionTest(unittest.TestCase):
    """Real Handler, real ThreadingHTTPServer, real config/secret — LAN mode
    with a password, in an isolated home and tickets dir."""

    @classmethod
    def setUpClass(cls):
        home = Path(tempfile.mkdtemp(prefix="wb108-home-"))
        (home / "tickets").mkdir()
        cls.password = "board-pw-108"
        (home / "config.json").write_text(json.dumps({
            "port": 0,
            "default_project": str(home / "projekt"),
            "projects": {"Harness": str(home / "projekt")},
            "lan": True,
            "password_hash": auth.hash_password(cls.password),
            "state_path": str(home / "state.json"),
            "agent_timeout_minutes": 5,
        }), encoding="utf-8")
        (home / "projekt").mkdir(exist_ok=True)
        # A LAN board is actually reached at its own IP — 127.0.0.1 is not a
        # private address. Use the machine's real LAN IP so these tests also
        # cover the guard's LAN-host branch, exactly as a phone would.
        import socket as _socket
        _s = _socket.socket(_socket.AF_INET, _socket.SOCK_DGRAM)
        try:
            _s.connect(("8.8.8.8", 80))
            lan_ip = _s.getsockname()[0]
        except OSError:
            lan_ip = "127.0.0.1"
        _s.close()
        cls.lan_ip = lan_ip
        cls.port = _free_port()

        cls.port = _free_port()
        cls.proc = subprocess.Popen(
            [sys.executable, str(HARNESS_SCRIPT), str(cls.port), str(home)],
            stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True,
            env=dict(os.environ, HOME=str(home)))
        cls._home = home

    @classmethod
    def tearDownClass(cls):
        if cls.proc:
            cls.proc.kill()
            cls.proc.wait()
        shutil.rmtree(cls._home, ignore_errors=True)

    def _fail_if_board_dead(self):
        if self.__class__.proc.poll() is not None:
            err = self.__class__.proc.stderr.read()
            raise AssertionError(f"board process died: {err[:800]!r}")

    # -- helpers ---------------------------------------------------------------

    def _login(self, password=None):
        s, b, cookie = _req(self.port, "/api/login",
                            data={"password": self.password if password is None
                                 else password},
                            headers=board_headers(self.lan_ip, self.port))
        return s, _as_json(s, b), cookie

    @classmethod
    def _session_cookie(cls, set_cookie):
        for part in (set_cookie or "").split(";"):
            if part.strip().startswith("werkbank_session="):
                return part.strip().split("=", 1)[1]
        return None

    def _authed(self, set_cookie):
        h = board_headers(self.lan_ip, self.port)
        tok = self._session_cookie(set_cookie)
        if tok:
            h["Cookie"] = f"werkbank_session={tok}"
        return h

    def _get(self, path, **headers):
        s, b, _ = _req(self.port, path, headers=headers or None)
        return s, _as_json(s, b)

    def _post(self, path, data=None, **headers):
        h = board_headers(self.lan_ip, self.port)
        h.pop("Cookie", None)
        h.update(headers)
        s, b, cookie = _req(self.port, path, data=data, headers=h)
        return s, _as_json(s, b), cookie

    # -- hole 1: the auth gate is in the request path ---------------------------

    def test_unauthenticated_requests_are_rejected(self):
        """A stranger without a cookie gets 401 on data and on browse. With
        the handler's auth check removed this would be 200 — the hole from
        the 2026-08-16 audit."""
        s, b = self._get("/api/tickets")
        self.assertEqual(s, 401)
        s, b = self._get("/api/browse")
        self.assertEqual(s, 401)

    def test_wrong_password_is_rejected(self):
        s, b, cookie = self._login(password="falsch")
        self.assertEqual(s, 401)
        self.assertIsNone(self._session_cookie(cookie))

    def test_login_granting_a_cookie_opens_the_board(self):
        s, b, cookie = self._login()
        self.assertEqual(s, 200)
        self.assertIsNotNone(self._session_cookie(cookie))
        s, b = self._get("/api/tickets", **self._authed(cookie))
        self.assertEqual(s, 200)
        self.assertIn("tickets", b)
        # the credential material never leaves the server (public_config)
        self.assertNotIn("password_hash", b.get("config", {}))

    # -- hole 2: the browse roots come from server.py ---------------------------

    def test_browse_outside_the_roots_is_refused(self):
        """A path outside home+projects must not return a listing. Widening
        the root list to `/` in server.py turns this 200 — the second hole."""
        s, b, cookie = self._login()
        s, b = self._get("/api/browse?path=" + str(tempfile.gettempdir()),
                         **self._authed(cookie))
        self.assertIn(s, (400, 403), b)
        self.assertNotIn(tempfile.gettempdir(), str(b))  # no path oracle

    def test_browse_inside_a_project_root_is_allowed(self):
        """The same route works inside a registered project: the boundary is
        a root list, not a hard refusal."""
        s, b, cookie = self._login()
        s, b = self._get(f"/api/browse?path={self.__class__._home / 'projekt'}",
                         **self._authed(cookie))
        self.assertEqual(s, 200, b)
        self.assertEqual(b["path"], str((self.__class__._home / "projekt").resolve()))

    def test_browse_without_a_path_lists_the_first_root(self):
        s, b, cookie = self._login()
        s, b = self._get("/api/browse", **self._authed(cookie))
        self.assertEqual(s, 200, b)
        self.assertTrue(b["path"].startswith(str(self.__class__._home)))

    # -- guard and routing wiring ------------------------------------------------

    def test_foreign_origin_write_is_refused_by_guard(self):
        """do_POST runs guard.check_write first (F1): a hostile page cannot
        drive a ticket through the real request path."""
        s, b, _ = self._post("/api/tickets", {"title": "Niemals"},
                             Origin="https://evil.example")
        self.assertEqual(s, 403)
        self.assertIn("Herkunft", str(b))

    def test_unauthenticated_write_is_rejected(self):
        s, b, _ = self._post("/api/tickets", {"title": "Ohne Anmeldung"})
        self.assertEqual(s, 401)

    def test_foreign_host_read_is_refused(self):
        """F2: the Host header must name the board (DNS-rebinding guard)."""
        h = board_headers(self.lan_ip, self.port)
        h["Host"] = f"evil.test:{self.port}"
        s, b = self._get("/api/tickets", **h)
        self.assertEqual(s, 403)

    def test_form_content_type_write_is_refused(self):
        """text/plain needs no preflight — the CSRF vector from F1."""
        s, b, _ = _req(self.port, "/api/tickets/WB-1", data=b"status=in_arbeit",
                       headers={"Content-Type": "text/plain",
                                "Host": "127.0.0.1:8765"})
        self.assertEqual(s, 403, _as_json(s, b))

    def test_core_routes_answer(self):
        """/ answers with the board page, an unknown API route is 404, and
        the defense-in-depth headers (WB-35) are present at the boundary."""
        s, raw, _ = _req(self.port, "/")
        self.assertEqual(s, 200)
        self.assertIn(b"werkbank", raw.lower())
        s, b = self._get("/api/definitely_not_there.txt")
        self.assertEqual(s, 404)

    # -- the write path reaches the store through the routing --------------------

    def test_ticket_crud_roundtrip(self):
        """POST /api/tickets, PATCH-style update, DELETE — through the real
        routing, into the isolated ticket store."""
        s, b, cookie = self._login()
        h = self._authed(cookie)
        s, b, _ = self._post("/api/tickets",
                             {"title": "Kompositionstest", "description": "WB-108"},
                             **h)
        self.assertEqual(s, 200, b)
        created = b["id"]
        try:
            s, b, _ = self._post(f"/api/tickets/{created}",
                                 {"status": "in_arbeit"}, **h)
            self.assertEqual(s, 200, b)
            self.assertEqual(b["status"], "in_arbeit")
            t = next(x for x in store.load_tickets(self.__class__._home / "tickets")
                     if x.id == created)
            self.assertEqual(t.status, "in_arbeit")
        finally:
            s, b, _ = _req(self.port, f"/api/tickets/{created}", data={},
                           headers=self._authed(cookie), method="DELETE")
            self.assertEqual(s, 200, _as_json(s, b))
            self.assertNotIn(created, [x.id for x in
                                       store.load_tickets(self.__class__._home / "tickets")])


if __name__ == "__main__":
    unittest.main()
